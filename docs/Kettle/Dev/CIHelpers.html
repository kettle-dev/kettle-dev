Overview

CI-related helper functions used by Rake tasks and release tooling.

This module only exposes module-functions (no instance state) and is
intentionally small so it can be required by both Rake tasks and the
kettle-release executable.

def ci_rate_claim_in_process
  @ci_rate_inproc_last = 0.0 unless defined?(@ci_rate_inproc_last)
  now = Time.now.to_f
  slot = [@ci_rate_inproc_last + CI_RATE_MIN_INTERVAL, now].max
  @ci_rate_inproc_last = slot
  wait = slot - now
  wait.positive? ? wait : 0.0
end

def ci_rate_state_path
  override = ENV["K_RELEASE_CI_RATE_STATE_FILE"]
  return override if override && !override.empty?

  base = ENV["XDG_STATE_HOME"]
  base = (base && !base.empty?) ? base : File.join(Dir.home, ".local", "state")
  File.join(base, "kettle-dev", "ci-rate-budget.json")
end

def claim_ci_rate_slot
  @ci_rate_lock.synchronize do
    path = ci_rate_state_path
    dir = File.dirname(path)
    begin
      FileUtils.mkdir_p(dir)
    rescue SystemCallError
      # No writable state dir: degrade to in-process throttling only.
      return ci_rate_claim_in_process
    end

    begin
      File.open(path, File::RDWR | File::CREAT, 0o644) do |f|
        f.flock(File::LOCK_EX)
        payload = f.size.zero? ? {} : JSON.parse(f.read)
        last = payload["next_slot"] || payload["last_request"] || 0.0
        now = Time.now.to_f
        # Absurdly future (clock skew) or ancient entries start fresh.
        last = 0.0 if !last.is_a?(Numeric) || (last - now).abs > 3600
        slot = [last + CI_RATE_MIN_INTERVAL, now].max
        f.rewind
        f.truncate(0)
        f.write(JSON.generate({"next_slot" => slot}))
        f.flush
        wait = slot - now
        return wait.positive? ? wait : 0.0
      end
    rescue SystemCallError, JSON::ParserError
      # Ledger unusable (read-only FS, broken JSON, no flock support):
      # degrade to in-process throttling rather than failing releases.
      return ci_rate_claim_in_process
    end
  end
end

def describe_workflow_triggers(triggers)
  case triggers
  when nil
    "missing `on:` triggers"
  when String, Symbol
    "triggers: #{triggers}"
  when Array
    "triggers: #{triggers.join(", ")}"
  when Hash
    parts = triggers.map do |key, value|
      if key.to_s == "push" && value.is_a?(Hash) && value["branches"]
        "push[branches: #{Array(value["branches"]).join(", ")}]"
      elsif key.to_s == "push" && value.is_a?(Hash) && value["branches-ignore"]
        "push[branches-ignore: #{Array(value["branches-ignore"]).join(", ")}]"
      else
        key.to_s
      end
    end
    "triggers: #{parts.join(", ")}"
  else
    "invalid `on:` triggers"
  end
end

def github_branch_patterns_match?(branch, patterns)
  matched = false
  Array(patterns).each do |raw_pattern|
    pattern = raw_pattern.to_s
    if pattern.start_with?("!")
      negated = pattern.sub(/\A!/, "")
      matched = false if github_glob_to_regexp(negated).match?(branch)
    elsif github_glob_to_regexp(pattern).match?(branch)
      matched = true
    end
  end
  matched
end

def github_commit_exists?(owner:, repo:, head_sha:, token: default_token, **options)
  return nil unless owner && repo && head_sha && !head_sha.empty?

  url = "https://api.github.com/repos/#{owner}/#{repo}/commits/#{URI.encode_www_form_component(head_sha)}"
  res, _redirects = github_request(url, token: token)
  return true if res.is_a?(Net::HTTPSuccess)
  return false if res.is_a?(Net::HTTPNotFound) || res.is_a?(Net::HTTPUnprocessableEntity)

  if res.is_a?(Net::HTTPUnauthorized)
    raise(
      Kettle::Dev::CIAuthError,
      "GitHub API rejected the request (401: invalid or revoked token) for #{url}. " \
        "Check GITHUB_TOKEN / GH_TOKEN."
    )
  elsif res.is_a?(Net::HTTPForbidden) && res["X-RateLimit-Remaining"].to_i.zero?
    reset_at = Time.at(res["X-RateLimit-Reset"].to_i).utc
    raise Kettle::Dev::CIRateLimitError.new(
      "GitHub API rate limit exceeded; resets at #{reset_at.strftime("%H:%M:%SZ")}",
      reset_at: reset_at
    )
  end
  nil
rescue Kettle::Dev::CIAuthError, Kettle::Dev::CIRateLimitError, Kettle::Dev::CIConfigurationError
  raise
rescue => e
  Kettle::Dev.debug_error(e, __method__)
  nil
end

def github_get_json(url, token:, on_redirect: nil, **options)
  res, redirects = github_request(url, token: token, on_redirect: on_redirect)
  if on_redirect.nil?
    permanent = redirects.find { |hop| PERMANENT_REDIRECT_CODES.include?(hop[:code]) }
    if permanent
      warn(
        "[kettle-dev] WARNING: GitHub API permanently redirected #{permanent[:from]} -> " \
          "#{redirects.last[:to]} (HTTP #{permanent[:code]}). Update your git remote URL to the canonical repository."
      )
    end
  end

  if res.is_a?(Net::HTTPSuccess)
    JSON.parse(res.body)
  elsif res.is_a?(Net::HTTPUnauthorized)
    # 401 is not retryable: an invalid or revoked token fails EVERY
    # request, so a monitor treating it as "no data yet" would tick
    # forever at 0/N. Abort with an actionable message.
    raise(
      Kettle::Dev::CIAuthError,
      "GitHub API rejected the request (401: invalid or revoked token) for #{url}. " \
        "Check GITHUB_TOKEN / GH_TOKEN."
    )
  elsif res.is_a?(Net::HTTPForbidden)
    if res["X-RateLimit-Remaining"].to_i.zero?
      # Rate limited: recoverable. Honor the reset time, then let the
      # caller retry the poll cycle.
      reset_at = Time.at(res["X-RateLimit-Reset"].to_i).utc
      raise Kettle::Dev::CIRateLimitError.new(
        "GitHub API rate limit exceeded; resets at #{reset_at.strftime("%H:%M:%SZ")}",
        reset_at: reset_at
      )
    end

    # Non-quota 403 (e.g. resource permissions): treat as no data, like
    # other transient responses.
    nil
  else
    nil
  end
end

def github_glob_to_regexp(pattern)
  escaped = pattern.to_s.gsub(/(\*\*|\*|\?|[^*?]+)/) do |token|
    case token
    when "**" then ".*"
    when "*" then "[^/]*"
    when "?" then "[^/]"
    else Regexp.escape(token)
    end
  end
  /\A#{escaped}\z/
end

.github_rate_limit_wait! ⇒ Object

Cross-process spacing, ticket-style: each caller claims the next
request slot under an exclusive flock on the ledger file, RELEASES the
lock, and only then sleeps out its wait. The critical section is a few
microseconds regardless of budget, so hammering claimers queue briefly
on the flock instead of each holding it for a whole interval — the old
in-lock sleep serialized the whole machine behind one interval per
claim, and a claimer wedged mid-sleep deadlocked every other process
(observed in CI: a spec-run worker held the flock for minutes while
its peers silently starved past the watchdog threshold).
The flock is released automatically if a process dies. A missing or
corrupt ledger starts fresh.

def github_rate_limit_wait!

def github_repo_info(owner:, repo:, token: default_token, **options)
  return unless owner && repo

  url = "https://api.github.com/repos/#{owner}/#{repo}"
  res, redirects = github_request(url, token: token)
  if res.is_a?(Net::HTTPSuccess)
    data = JSON.parse(res.body)
    {
      "data" => data,
      "full_name" => data["full_name"],
      "default_branch" => data["default_branch"],
      "archived" => data["archived"],
      "redirected" => redirects.any? { |hop| PERMANENT_REDIRECT_CODES.include?(hop[:code]) },
      "redirects" => redirects
    }
  elsif res.is_a?(Net::HTTPNotFound)
    raise(
      Kettle::Dev::CIConfigurationError,
      "GitHub repository #{owner}/#{repo} not found (HTTP 404). " \
        "Check the git remote URL and ensure GITHUB_TOKEN / GH_TOKEN has access to this repository."
    )
  elsif res.is_a?(Net::HTTPUnauthorized)
    raise(
      Kettle::Dev::CIAuthError,
      "GitHub API rejected the request (401: invalid or revoked token) for #{url}. " \
        "Check GITHUB_TOKEN / GH_TOKEN."

def github_request(url, token:, on_redirect: nil, max_redirects: MAX_GITHUB_REDIRECTS)
  current_url = url.to_s
  initial_host = URI(current_url).hostname
  redirects = []

  (max_redirects + 1).times do
    github_rate_limit_wait!
    uri = URI(current_url)
    req = Net::HTTP::Get.new(uri)
    req["User-Agent"] = "kettle-dev/ci-helpers"
    if token && !token.empty? && uri.hostname == initial_host
      req["Authorization"] = "token #{token}"
    end
    res = Net::HTTP.start(uri.hostname, uri.port, use_ssl: true) { |http| http.request(req) }

    # Dispatch on is_a? rather than `case`/===: verifying doubles
    # (instance_double(Net::HTTPOK)) used in specs do not reliably match
    # `Net::HTTPSuccess === res`, which silently degraded every stubbed
    # response to "no data".
    return [res, redirects] if res.is_a?(Net::HTTPSuccess)

    if res.is_a?(Net::HTTPRedirection)
      location = res["location"] || res["Location"]
      return [res, redirects] if location.nil? || location.empty?

      next_url = URI.join(current_url, location).to_s
      hop = {from: current_url, to: next_url, code: res.code.to_i}
      redirects << hop
      on_redirect&.call(hop)
      current_url = next_url
      next
    end

    return [res, redirects]
  end

  raise(
    Kettle::Dev::CIConfigurationError,
    "GitHub API exceeded #{max_redirects} redirects starting from #{url}."
  )
end

    "id" => pipe["id"],

def latest_run(owner:, repo:, workflow_file:, branch: nil, token: default_token, require_head: false, head_sha: nil)
  return unless owner && repo

  b = branch || current_branch
  return unless b

  # Scope to the exact commit SHA when available to avoid picking up a previous run on the same branch.
  sha = head_sha || current_head_sha
  data = github_get_json(
    "https://api.github.com/repos/#{owner}/#{repo}/actions/workflows/#{workflow_file}/runs?branch=#{URI.encode_www_form_component(b)}&per_page=100",
    token: token
  )
  return unless data

  runs = Array(data["workflow_runs"]) || []
  # Match by head_sha first. There may be multiple builds for one commit;
  # always select the most recent build rather than trusting API order.
  run = if sha
    match = preferred_head_run(runs.select { |r| r["head_sha"] == sha })
    match ||= latest_repository_workflow_run(owner: owner, repo: repo, workflow_file: workflow_file, branch: b, head_sha: sha, token: token)
    require_head ? match : (match || runs.first)
  else
    runs.first unless require_head
  end
  return unless run

  {
    "status" => run["status"],
    "conclusion" => run["conclusion"],
    "html_url" => run["html_url"],
    "id" => run["id"],
    "head_sha" => run["head_sha"]
  }
rescue Kettle::Dev::CIAuthError, Kettle::Dev::CIRateLimitError, Kettle::Dev::CIConfigurationError
  # Auth and configuration failures are fatal; rate limits are recoverable
  # but must reach the monitor (which parks until the reset) — swallowing
  # any of them turns the poll into an infinite 0/N tick.
  raise
rescue => e
  Kettle::Dev.debug_error(e, __method__)
  nil
end

def parse_hosted_repo(url, host)
  return unless url

  if url =~ %r{\Agit@#{Regexp.escape(host)}:(.+?)/(.+?)(?:\.git)?\z}

def push_trigger_matches_branch?(triggers, branch)
  case triggers
  when String, Symbol
    triggers.to_s == "push"
  when Array
    triggers.any? { |t| t.to_s == "push" }
  when Hash
    return false unless triggers.key?("push") || triggers.key?(:push)

    push_cfg = triggers.key?("push") ? triggers["push"] : triggers[:push]
    return true if push_cfg.nil?
    return true unless push_cfg.is_a?(Hash)

    branches = push_cfg["branches"]
    branches_ignore = push_cfg["branches-ignore"]
    tags = push_cfg["tags"]
    tags_ignore = push_cfg["tags-ignore"]

    return false if branches.nil? && branches_ignore.nil? && (!tags.nil? || !tags_ignore.nil?)
    return github_branch_patterns_match?(branch, Array(branches)) unless branches.nil?
    return !github_branch_patterns_match?(branch, Array(branches_ignore)) unless branches_ignore.nil?

    true
  else
    false
  end
end

def release_jobs_verdict(jobs, run_completed: false)
  relevant = Array(jobs).reject { |job| slow_job?(job["name"]) }
  return [nil, nil] if relevant.empty?

  (all_done && !truncated) ? [:passed, nil] : [nil, nil]

  trigger_present?(triggers, "pull_request") || trigger_present?(triggers, "workflow_dispatch")
end

def workflows_list(root = project_root)
  workflows_dir = File.join(root, ".github", "workflows")
  files = if Dir.exist?(workflows_dir)
    Dir[File.join(workflows_dir, "*.yml")] + Dir[File.join(workflows_dir, "*.yaml")]
  else
    []
  end
  basenames = files.map { |p| File.basename(p) }
  basenames = basenames.uniq - exclusions
  basenames.reject { |name| slow_workflow?(name) }.sort
end
